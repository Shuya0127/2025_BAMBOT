#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>
#include <TinyGPSPlus.h> 

// === 1. MACアドレス定義 ===
// 受信機ESP32のMACアドレス (GPSデータ送信先 / コマンド受信元)
// 🚨 あなたの受信機ESP32のMACアドレスに置き換えてください！
uint8_t peerMac_Receiver[] = {0x9C, 0x9C, 0x1F, 0xCF, 0xE9, 0x40}; 

// === 2. LEDピン定義 ===
#define LED_PIN 2 // GPIO2をLED制御に使用
#define IN1 18
#define IN2 19
#define ENA 5
#define IN3 26//9番ピンを出力ピンIN3と定義してます。
#define IN4 25//8番ピンを出力ピンIN4と定義してます。
#define IN5 14//9番ピンを出力ピンIN3と定義してます。
#define IN6 12//8番ピンを出力ピンIN4と定義してます。
#define ENB 27
#define SPEEDA 220
#define SPEEDB 90

// === 3. GPSモジュール用ピン定義 (既存) ===
#define GPS_RX_PIN 16 
#define GPS_TX_PIN 17 
#define GPS_BAUD 9600 

HardwareSerial SerialGPS(2); 
TinyGPSPlus gps;

// === 4. データ構造体定義 ===

// 制御コマンド受信用の構造体 (受信機から送られてくる)
typedef struct struct_command {
  char type[10];      // "LED_CMD"
  bool led_state;     // true: ON, false: OFF
} struct_command;

// GPS送信データ構造体 (既存)
typedef struct struct_message {
  bool is_valid_gps;    
  double latitude;    
  double longitude;   
  int satellites;     
} struct_message;

struct_message myData;

// === 5. コールバック関数定義 ===

// 送信完了コールバック (前回修正済み)
void OnDataSent(const wifi_tx_info_t *info, esp_now_send_status_t status) {
  Serial.print("GPSデータ送信結果: ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "成功" : "失敗");
}

// 受信コールバック (今回修正済み)
void OnDataRecv(const esp_now_recv_info_t *info, const uint8_t *incomingData, int len) {
  struct_command commandData;
  
  // MACアドレスが必要な場合は info->src_addr でアクセス可能
  // (今回はMACアドレスは使用しないためロジックはそのまま)

  if (len == sizeof(commandData)) {
    memcpy(&commandData, incomingData, sizeof(commandData));

    // データタイプが "LED_CMD" であることを確認
    if (strcmp(commandData.type, "LED_CMD") == 0) {
      if (commandData.led_state) {
        digitalWrite(LED_PIN, HIGH); // LED点灯
        digitalWrite(IN1,HIGH);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN2,LOW);      //正転
        
        //for(int i=255;i>=SPEEDA;i--){
          analogWrite(ENA,255);
          //delay(5);
        //}
        analogWrite(ENB,255);
        digitalWrite(IN3,HIGH);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN4,LOW);
        delay(100);
        digitalWrite(IN3,LOW);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN4,LOW);
        delay(100);
        digitalWrite(IN4,HIGH);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN3,LOW);
        delay(100);
        analogWrite(ENB,0);
        delay(100);
        digitalWrite(IN5,HIGH);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN6,LOW);
        delay(100);
        digitalWrite(IN5,LOW);     //どちらかがHIGHでモータが周ります。
        digitalWrite(IN6,LOW);
        Serial.println(">>> コマンド受信: LED ON");
      } else {
        digitalWrite(LED_PIN, LOW);  // LED消灯
        digitalWrite(IN1,LOW);      //停止（惰性回転）
        digitalWrite(IN2,LOW);
        digitalWrite(IN3,LOW);      //停止（惰性回転）
        digitalWrite(IN4,LOW);
        digitalWrite(IN5,LOW);      //停止（惰性回転）
        digitalWrite(IN6,LOW);
        analogWrite(ENA,0);
        analogWrite(ENB,0);
        Serial.println(">>> コマンド受信: LED OFF");
      }
    }
  }
}

// GPSデータ処理に集中するためのヘルパー関数 (既存)
void smartDelay(unsigned long ms) {
  unsigned long start = millis();
  do {
    while (SerialGPS.available())
      gps.encode(SerialGPS.read());
    
  } while (millis() - start < ms);
}


void setup() {
  Serial.begin(115200);
  Serial.println("\n=== ESP32 GPS/ESP-NOW 送信機 起動 (双方向モード) ===");

  //モーターピンを初期化
  pinMode(IN1,OUTPUT);
  pinMode(IN2,OUTPUT);
  pinMode(IN3,OUTPUT);
  pinMode(IN4,OUTPUT);
  pinMode(IN5,OUTPUT);
  pinMode(IN6,OUTPUT);
  pinMode(ENA,OUTPUT);
  pinMode(ENB,OUTPUT);
  // LEDピンを初期化
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW); // 最初は消灯

  digitalWrite(IN1,LOW);      //最初は停止（惰性回転）
  digitalWrite(IN2,LOW);
  digitalWrite(IN3,LOW);      //最初は停止（惰性回転）
  digitalWrite(IN4,LOW);
  digitalWrite(IN5,LOW);      //最初は停止（惰性回転）
  digitalWrite(IN6,LOW);
  SerialGPS.begin(GPS_BAUD, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  
  WiFi.mode(WIFI_STA);
  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW初期化失敗");
    return;
  }

  // 送受信両方のコールバックを登録
  esp_now_register_send_cb(OnDataSent);
  esp_now_register_recv_cb(OnDataRecv); 

  // ピアの追加
  esp_now_peer_info_t peerInfo;
  memset(&peerInfo, 0, sizeof(peerInfo)); 
  memcpy(peerInfo.peer_addr, peerMac_Receiver, 6);
  peerInfo.channel = 0;
  peerInfo.encrypt = false;
  
  if (esp_now_add_peer(&peerInfo) != ESP_OK) {
    Serial.println("ピア追加失敗");
  } else {
    Serial.println("システム準備完了。");
  }
}

void loop() {
  // GPSデータ処理（5秒間隔のメインロジック） (既存)
  smartDelay(100); 

  static unsigned long last_gps_send_time = 0;
  if (millis() - last_gps_send_time >= 5000) {

    // GPSデータ処理 
    if (gps.location.isUpdated()) {
      myData.is_valid_gps = true;
      myData.latitude = gps.location.lat();
      myData.longitude = gps.location.lng();
      myData.satellites = gps.satellites.isValid() ? gps.satellites.value() : 0;
      Serial.println("GPSデータ送信 (有効)");
    } else {
      myData.is_valid_gps = false;
      myData.latitude = 0.0;
      myData.longitude = 0.0;
      myData.satellites = 0;
      Serial.println("GPSデータ送信 (無効)");
    }

    // GPS Fixの有無にかかわらず、定期的にデータを送信
    esp_err_t result = esp_now_send(peerMac_Receiver, (uint8_t *)&myData, sizeof(myData));

    last_gps_send_time = millis(); 
  }
}
